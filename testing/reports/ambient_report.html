suppressPackageStartupMessages({
  library(data.table)
  library(magrittr)
  library(Matrix)
  library(ggplot2)
  library(patchwork)
  library(strex)
})

# Discover per-sample files written by DECONTX into the Nextflow work dir
qc_fs      <- list.files(".", pattern = "^barcodes_qc_metrics_.+\\.csv\\.gz$", full.names = TRUE)
bcs_fs     <- list.files(".", pattern = "^cell_barcodes_.+\\.csv$",            full.names = TRUE)
dcx_fs     <- list.files(".", pattern = "^dcx_params_.+\\.csv$",               full.names = TRUE)
summary_fs <- list.files(".", pattern = "^dcx_summary_.+\\.csv$",              full.names = TRUE)

# Extract sample IDs from filenames
.sid <- function(fs, prefix, suffix) {
  sub(suffix, "", sub(paste0("^.*/", prefix), "", fs))
}

s_lvls <- .sid(summary_fs, "dcx_summary_", "\\.csv$")
names(qc_fs)      <- .sid(qc_fs,      "barcodes_qc_metrics_", "\\.csv\\.gz$")
names(bcs_fs)     <- .sid(bcs_fs,     "cell_barcodes_",       "\\.csv$")
names(dcx_fs)     <- .sid(dcx_fs,     "dcx_params_",          "\\.csv$")
names(summary_fs) <- .sid(summary_fs, "dcx_summary_",         "\\.csv$")

# Align to same sample order
qc_fs      <- qc_fs[s_lvls]
bcs_fs     <- bcs_fs[s_lvls]
dcx_fs     <- dcx_fs[s_lvls]
summary_fs <- summary_fs[s_lvls]

# Load data
summary_dt <- lapply(summary_fs, fread) %>% rbindlist()
ok_bcs_ls  <- lapply(bcs_fs,     function(f) fread(f, header = FALSE)$V1)
dcx_ls     <- lapply(dcx_fs,     fread)

# Filter QC metrics to barcodes with >= 10 total pre-UMIs (mirrors scprocess get_usa_dt)
usa_dt_ls  <- lapply(qc_fs, function(f) {
  dt <- fread(f)
  dt[pre_S + pre_U + pre_A >= 10]
})

n_samples  <- length(s_lvls)
chunk_size <- 10
idx_ls     <- seq_along(s_lvls)
chunks_ls  <- split(idx_ls, ceiling(idx_ls / chunk_size))

Spliced % vs UMIs

Relationship between total UMIs and spliced read percentage, before and after decontX. Ambient RNA is enriched for spliced reads, so effective ambient removal shifts the spliced % of cells downward. Barcodes are coloured by whether they were called as cells (grey), used as ambient background (red), or excluded entirely (dark grey).
